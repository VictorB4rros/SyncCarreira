%% Lista de Símbolos
\begin{simbolos}
  \item [$\Delta$] Delta (variação ou diferença em processos)
  \item [$\sum$] Somatório (usado na análise de dados e métricas)
  \item [$\mathbb{R}$] Conjunto dos números reais (análise financeira)
  \item [$\%$] Porcentagem (indicadores de desempenho e métricas)
  \item [$\S$] Seção (referência a artigos da LGPD ou normas)
  \item [$\copyright$] Copyright (propriedade intelectual do software)
  \item [$\registered$] Marca registrada (tecnologias licenciadas)
\end{simbolos}
